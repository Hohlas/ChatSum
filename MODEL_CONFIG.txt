# Model settings, tracked in git. One push to main updates VPS and Actions, no Secrets/Variables change.
# File wins over env; env is fallback only.
# private.txt holds ONLY keys (TELEGRAM_*, GOOGLE_API_KEY*, OPENROUTER_API_KEY*, OPENCODE_GO_API_KEY).
# All MODEL / TEMPERATURE / REASONING_EFFORT / CHUNK_MAX_CHARS live here, per provider.
# REASONING_EFFORT: none low medium high.
# LLM_PROVIDER: google | openrouter | opencode | nvidia (default google).
# Выбор по номерам — команда /model в боте (пишет сюда же).

# ═══ КАК УСТРОЕН ФАЙЛ ═══════════════════════════════════════════════════════
# Блок на провайдера: строка <ПРЕФИКС>_MODEL= (какая модель у этого
# провайдера) и рядом его параметры — TEMPERATURE / REASONING_EFFORT /
# CHUNK_MAX_CHARS. В конце — раздел «АКТИВНАЯ МОДЕЛЬ»: LLM_PROVIDER=<кто
# активен> и повтор строки модели выбранного провайдера. Обе копии модели
# пишет бот командой /model и держит одинаковыми. Если один ключ встретится
# дважды с разными значениями, действует последняя строка — бот напишет об
# этом предупреждение в лог. Длина блока/раздела роли не играет: блок
# находится по заголовку, а не по номеру строки.

# ═══ GOOGLE / GEMINI ════════════════════════════════════════════════════════
GEMINI_MODEL=gemini-3.8-flash
GEMINI_TEMPERATURE=0               # 0 = детерминированный (рекомендуется), 0.3 = вариативный
GEMINI_REASONING_EFFORT=high     # none low medium high (none — детерминизм, medium — глубже анализ но нестабилен)
GEMINI_CHUNK_MAX_CHARS=40000      # ~15K токенов входа (в JSON ~2.6 знака на токен)

# ═══ OPENROUTER ═════════════════════════════════════════════════════════════
# Потолок ответа Qwen в коде — 30 000 токенов: OpenRouter резервирует под запрос
# потолок ответа по цене выхода ($0.47/1M → 65 536 ≈ $0.031) и при тощем балансе
# отдаёт 402 «can only afford N tokens» ещё до отправки модели.
#   OPENROUTER_MODEL=qwen/qwen3.8-flash                      OPENROUTER_CHUNK_MAX_CHARS=80000    # 1 000 000 / 30 000
#   OPENROUTER_MODEL=nvidia/nemotron-3-ultra-550b-a55b:free  OPENROUTER_CHUNK_MAX_CHARS=100000   # 1 000 000 / 65 536
#   OPENROUTER_MODEL=nvidia/nemotron-3-super-120b-a12b:free  OPENROUTER_CHUNK_MAX_CHARS=100000   # 262 144 / 235 929
#   OPENROUTER_MODEL=google/gemma-4-31b-it:free              OPENROUTER_CHUNK_MAX_CHARS=60000    # 262 144 / 32 768 (slug строго с '-it')
#   OPENROUTER_MODEL=deepseek/deepseek-v4.1-flash            OPENROUTER_CHUNK_MAX_CHARS=60000    # 1 048 576 / 131 072; в коде профиль-дефолт 128 000 / 16 384
OPENROUTER_MODEL=qwen/qwen3.8-flash
OPENROUTER_TEMPERATURE=0
OPENROUTER_REASONING_EFFORT=medium
OPENROUTER_CHUNK_MAX_CHARS=80000   # ~30K токенов входа; потолок ответа в коде 30 000

# ═══ OPENCODE GO ════════════════════════════════════════════════════════════
# OpenCode Go (подписка opencode.ai), ключ OPENCODE_GO_API_KEY. DeepSeek думает
# по умолчанию; шлюз принимает reasoning_effort (как OpenAI), но в доках его нет.
# Контекст DeepSeek V4.1 Flash — 1 048 576 токенов (каталог модели; шлюз принял
# 320 799 токенов входа живьём). 100 000 знаков JSON = ~38 000 токенов входа;
# живой замер 28.10: ответ 2 114 токенов (рассуждения 1 422), finish=stop, 23 с.
# Часы пик по Москве (UTC+3, только пн–пт):
# 04:00–07:00 МСК
# 09:00–13:00 МСК
OPENCODE_MODEL=deepseek-v4.1-flash
OPENCODE_TEMPERATURE=0
OPENCODE_REASONING_EFFORT=low
OPENCODE_CHUNK_MAX_CHARS=100000

# ═══ NVIDIA ═════════════════════════════════════════════════════════════════
NVIDIA_MODEL=deepseek-ai/deepseek-v4.1-flash
NVIDIA_TEMPERATURE=0
NVIDIA_REASONING_EFFORT=low
NVIDIA_CHUNK_MAX_CHARS=40000

# ═══ АКТИВНАЯ МОДЕЛЬ (строки пишет бот командой /model) ═════════════════════
# LLM_PROVIDER: google | openrouter | opencode | nvidia (default google).
# /model даёт три номера: openrouter (Qwen), opencode (DeepSeek), google (Gemini).
# NVIDIA выбирается только через /set_provider nvidia.
# Модель внизу — повтор строки модели активного провайдера. Читается
# последней, поэтому именно она решает, какая модель пойдёт в анализ.

LLM_PROVIDER=opencode

OPENCODE_MODEL=deepseek-v4.1-flash
